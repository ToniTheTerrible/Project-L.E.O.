def run_splash_screen():
    """Renders the complex Tactical Green HUD splash screen."""
    root = tk.Tk()
    root.overrideredirect(True) # Frameless window
    root.attributes("-topmost", True) # Keep on top during boot

    # Center window
    w, h = 460, 460
    sw = root.winfo_screenwidth()
    sh = root.winfo_screenheight()
    x, y = (sw - w) // 2, (sh - h) // 2
    root.geometry(f"{w}x{h}+{x}+{y}")

    bg_color = "#020B05"
    canvas = tk.Canvas(root, width=w, height=h, bg=bg_color, highlightthickness=0)
    canvas.pack(fill="both", expand=True)

    cx, cy = w // 2, h // 2

    # Color Palette: Tactical Green
    GREEN_BRIGHT = "#00FF41"
    GREEN_MID    = "#00B33C"
    GREEN_DARK   = "#004D1A"
    GREEN_TEXT   = "#55FF88"

    # 1. Corner Reticle Brackets
    b_len = 25
    b_pad = 20
    # Top-Left
    canvas.create_line(b_pad, b_pad, b_pad + b_len, b_pad, fill=GREEN_MID, width=2)
    canvas.create_line(b_pad, b_pad, b_pad, b_pad + b_len, fill=GREEN_MID, width=2)
    # Top-Right
    canvas.create_line(w - b_pad, b_pad, w - b_pad - b_len, b_pad, fill=GREEN_MID, width=2)
    canvas.create_line(w - b_pad, b_pad, w - b_pad, b_pad + b_len, fill=GREEN_MID, width=2)
    # Bottom-Left
    canvas.create_line(b_pad, h - b_pad, b_pad + b_len, h - b_pad, fill=GREEN_MID, width=2)
    canvas.create_line(b_pad, h - b_pad, b_pad, h - b_pad - b_len, fill=GREEN_MID, width=2)
    # Bottom-Right
    canvas.create_line(w - b_pad, h - b_pad, w - b_pad - b_len, h - b_pad, fill=GREEN_MID, width=2)
    canvas.create_line(w - b_pad, h - b_pad, w - b_pad, h - b_pad - b_len, fill=GREEN_MID, width=2)

    # 2. Crosshair Center Guides
    canvas.create_line(cx - 190, cy, cx - 165, cy, fill=GREEN_DARK, width=1)
    canvas.create_line(cx + 165, cy, cx + 190, cy, fill=GREEN_DARK, width=1)
    canvas.create_line(cx, cy - 190, cx, cy - 165, fill=GREEN_DARK, width=1)
    canvas.create_line(cx, cy + 165, cx, cy + 190, fill=GREEN_DARK, width=1)

    # 3. Rotating Arc Ring Definitions (radius, width, color, arcs, speed)
    rings = [
        # Outer heavy ring
        {"r": 175, "width": 3, "color": GREEN_BRIGHT, "arcs": [(0, 50), (90, 50), (180, 50), (270, 50)], "speed": 1.5},
        # Outer thin segmented ring
        {"r": 160, "width": 1, "color": GREEN_MID, "arcs": [(10, 30), (50, 15), (100, 45), (160, 20), (200, 40), (270, 30), (320, 25)], "speed": -2.2},
        # Tech block ring
        {"r": 125, "width": 5, "color": GREEN_BRIGHT, "arcs": [(0, 70), (120, 70), (240, 70)], "speed": 3.0},
        # Middle dashed ring
        {"r": 100, "width": 2, "color": GREEN_MID, "arcs": [(0, 110), (130, 110), (260, 80)], "speed": -4.0},
        # Inner fine ring
        {"r": 75,  "width": 2, "color": GREEN_BRIGHT, "arcs": [(15, 60), (105, 60), (195, 60), (285, 60)], "speed": 4.5},
        # Static core boundary
        {"r": 55,  "width": 1, "color": GREEN_DARK, "arcs": [(0, 360)], "speed": 0}
    ]

    arc_items = []
    for r_info in rings:
        r = r_info["r"]
        bbox = (cx - r, cy - r, cx + r, cy + r)
        group = []
        for start, extent in r_info["arcs"]:
            arc_id = canvas.create_arc(
                bbox, start=start, extent=extent, style="arc",
                outline=r_info["color"], width=r_info["width"]
            )
            group.append((arc_id, start))
        arc_items.append((group, r_info["speed"]))

    # 4. Outer Compass Tick Marks (36 Radial Teeth)
    tick_items = []
    num_ticks = 36
    r_tick_in = 142
    r_tick_out = 152
    for i in range(num_ticks):
        angle_deg = i * (360 / num_ticks)
        rad = math.radians(angle_deg)
        x1 = cx + r_tick_in * math.cos(rad)
        y1 = cy + r_tick_in * math.sin(rad)
        x2 = cx + r_tick_out * math.cos(rad)
        y2 = cy + r_tick_out * math.sin(rad)
        line_id = canvas.create_line(x1, y1, x2, y2, fill=GREEN_DARK, width=1)
        tick_items.append((line_id, angle_deg))

    # 5. Counter-Rotating Orbit Dots (24 dots)
    num_dots = 24
    r_dots = 112
    dot_items = []
    for i in range(num_dots):
        angle_deg = i * (360 / num_dots)
        rad = math.radians(angle_deg)
        dx = cx + r_dots * math.cos(rad)
        dy = cy + r_dots * math.sin(rad)
        dot_id = canvas.create_oval(dx - 1.5, dy - 1.5, dx + 1.5, dy + 1.5, fill=GREEN_MID, outline="")
        dot_items.append((dot_id, angle_deg))

    # Center Text
    canvas.create_text(cx, cy - 8, text="L.E.O.", font=("Consolas", 32, "bold"), fill=GREEN_BRIGHT)
    canvas.create_text(cx, cy + 22, text="TACTICAL HUD", font=("Consolas", 9, "bold"), fill=GREEN_MID)

    # Status Message Label
    status_item = canvas.create_text(cx, h - 30, text=status_text, font=("Consolas", 9), fill=GREEN_TEXT)

    # Start backend thread
    threading.Thread(target=load_leo_backend, daemon=True).start()

    def animate():
        if is_loaded:
            root.destroy()
            return

        # 1. Rotate Arc Rings
        for group, speed in arc_items:
            if speed == 0:
                continue
            for i, (arc_id, current_start) in enumerate(group):
                new_start = (current_start + speed) % 360
                canvas.itemconfig(arc_id, start=new_start)
                group[i] = (arc_id, new_start)

        # 2. Rotate Compass Ticks
        tick_speed = 0.8
        for i, (line_id, current_angle) in enumerate(tick_items):
            new_angle = (current_angle + tick_speed) % 360
            rad = math.radians(new_angle)
            x1 = cx + r_tick_in * math.cos(rad)
            y1 = cy + r_tick_in * math.sin(rad)
            x2 = cx + r_tick_out * math.cos(rad)
            y2 = cy + r_tick_out * math.sin(rad)
            canvas.coords(line_id, x1, y1, x2, y2)
            tick_items[i] = (line_id, new_angle)

        # 3. Rotate Orbit Dots
        dot_speed = -2.0
        for i, (dot_id, current_angle) in enumerate(dot_items):
            new_angle = (current_angle + dot_speed) % 360
            rad = math.radians(new_angle)
            dx = cx + r_dots * math.cos(rad)
            dy = cy + r_dots * math.sin(rad)
            canvas.coords(dot_id, dx - 1.5, dy - 1.5, dx + 1.5, dy + 1.5)
            dot_items[i] = (dot_id, new_angle)

        # Update status text
        canvas.itemconfig(status_item, text=status_text)
        root.after(20, animate)

    animate()
    root.mainloop()

# Run Intro Animation & Background Loading
run_splash_screen()

print("====================================================")
print("  [ L.E.O. TACTICAL HUD ONLINE ]")
print("  [ SYSTEM STATUS: ONLINE - BACKEND: GROQ GPT-OSS ]")
print("====================================================\n")

while True:
    try:
        user_input = input("> ")
    except (EOFError, KeyboardInterrupt):
        print("\n[ LEO OFFLINE ]")
        break

    if not user_input.strip():
        continue
    if user_input.strip().lower() in ("exit", "quit"):
        print("[ LEO OFFLINE ]")
        break

    # Prevent Groq 8,000 TPM Rate Limit by capping chat history
    if len(interpreter.messages) > 6:
        interpreter.messages = interpreter.messages[-6:]

    try:
        has_printed = False
        for chunk in interpreter.chat(user_input, display=False, stream=True):
            if isinstance(chunk, dict) and chunk.get("type") == "message":
                content = chunk.get("content")
                if content:
                    print(str(content), end="", flush=True)
                    has_printed = True

        if not has_printed:
            print("[ Task executed successfully. ]")
        else:
            print()

    except Exception as e:
        if "Rate limit" in str(e) or "429" in str(e):
            print("\n[ LEO: Groq rate limit hit. Please wait ~15 seconds before trying again. ]")
        else:
            print(f"\n[ LEO Error: {e} ]")
